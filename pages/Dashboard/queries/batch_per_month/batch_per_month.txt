SELECT *
FROM (
    SELECT
        DATE_FORMAT(calendar.calendar_month, '%Y-%m') AS month,
        COUNT(batch.batch_id) AS total_batches
    FROM (
        SELECT
            DATE_FORMAT(DATE_ADD(min_date, INTERVAL n MONTH), '%Y-%m-01') AS calendar_month
        FROM (
            SELECT 0 n UNION ALL SELECT 1 UNION ALL SELECT 2 UNION ALL SELECT 3 UNION ALL SELECT 4 UNION ALL
            SELECT 5 UNION ALL SELECT 6 UNION ALL SELECT 7 UNION ALL SELECT 8 UNION ALL SELECT 9 UNION ALL
            SELECT 10 UNION ALL SELECT 11 UNION ALL SELECT 12 UNION ALL SELECT 13 UNION ALL SELECT 14 UNION ALL
            SELECT 15 UNION ALL SELECT 16 UNION ALL SELECT 17 UNION ALL SELECT 18 UNION ALL SELECT 19 UNION ALL
            SELECT 20 UNION ALL SELECT 21 UNION ALL SELECT 22 UNION ALL SELECT 23 UNION ALL SELECT 24 UNION ALL
            SELECT 25 UNION ALL SELECT 26 UNION ALL SELECT 27 UNION ALL SELECT 28 UNION ALL SELECT 29 UNION ALL
            SELECT 30
        ) numbers
        CROSS JOIN (SELECT DATE_FORMAT(MIN(submission_date), '%Y-%m-01') AS min_date FROM batch) base
        WHERE DATE_ADD(min_date, INTERVAL n MONTH) <= (
            SELECT DATE_FORMAT(MAX(submission_date), '%Y-%m-01') FROM batch
        )
    ) calendar
    LEFT JOIN batch ON DATE_FORMAT(batch.submission_date, '%Y-%m-01') = calendar.calendar_month
    GROUP BY calendar.calendar_month
    ORDER BY calendar.calendar_month DESC
    LIMIT 12
) AS latest_months
ORDER BY latest_months.month ASC;
